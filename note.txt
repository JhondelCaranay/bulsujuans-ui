
### build image
docker build -t jhondel/bulsujuans-frontend-app:latest .
OR WITH ENV
docker build -t jhondel/bulsujuans-frontend-app:latest \
  --build-arg NEXT_PUBLIC_BACKEND_URL=http://localhost:4000 \
  --build-arg NODE_ENV=production \
  .


docker images
docker rmi bulsujuans-frontend-app:latest

### run container
docker run -d -p 4000:4000 --name bulsujuans-frontend-app-container --env-file .env jhondel/bulsujuans-frontend-app:latest
docker ps
docker rm bulsujuans-frontend-app-container


### docker compose
# Start / update containers
docker compose up -d

# Stop containers
docker compose down

# Rebuild (local dev only)
docker compose up -d --build

# View logs
docker compose logs -f
